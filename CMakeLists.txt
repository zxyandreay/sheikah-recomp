cmake_minimum_required(VERSION 3.22)
project(SheikahNativeProbe VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(sheikah_platform_probe STATIC native/src/platform_probe.cpp)
target_include_directories(sheikah_platform_probe PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/native/include)
if(MSVC)
    target_compile_options(sheikah_platform_probe PRIVATE /W4)
else()
    target_compile_options(sheikah_platform_probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(sheikah_platform_test native/tests/platform_probe_test.cpp)
    target_link_libraries(sheikah_platform_test PRIVATE sheikah_platform_probe)
    add_test(NAME native_platform_probe COMMAND sheikah_platform_test)
endif()
